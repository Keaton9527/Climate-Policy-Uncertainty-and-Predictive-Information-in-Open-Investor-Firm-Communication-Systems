import yaml
import pandas as pd
import numpy as np
import torch
from transformers import BertTokenizerFast
from transformers import AutoModelForSequenceClassification
from transformers import TextClassificationPipeline
from tqdm import tqdm
from snownlp import SnowNLP
import os

def load_model(model_path):
    # 检查是否有可用的GPU
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"Using device: {device}")
    
    model = AutoModelForSequenceClassification.from_pretrained(
        model_path,
        output_attentions=True,
        attn_implementation="eager"
    ).to(device)  # 将模型移动到GPU
    
    tokenizer = BertTokenizerFast.from_pretrained(model_path)
    return model, tokenizer, device

def load_config(config_path='code/src/bertConfig.yaml'):
    with open(config_path, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)
    return config

def get_score(label):
    score = {}
    for dct in (label[0]):
        # 处理MacBERT的默认标签输出
        if dct['label'] == 'LABEL_0':
            score['Negative'] = dct['score']
        elif dct['label'] == 'LABEL_1':
            score['Neutral'] = dct['score']
        elif dct['label'] == 'LABEL_2':
            score['Positive'] = dct['score']
        # 处理其他模型的标签输出
        elif dct['label'] == 'Positive':
            score['Positive'] = dct['score']
        elif dct['label'] == 'Negative':
            score['Negative'] = dct['score']
        elif dct['label'] == 'Neutral':
            score['Neutral'] = dct['score']
    return score

def load_data(data_path):
    df = pd.read_csv(data_path)
    # 剔除text列中的缺失值
    df = df.dropna(subset=['text'])
    print(f"原始数据量: {len(df)}")
    print(f"剔除缺失值后数据量: {len(df)}")
    return df

def process_batch(texts, PipelineInterface, batch_size=32):
    """批量处理文本"""
    results = []
    for i in range(0, len(texts), batch_size):
        batch = texts[i:i + batch_size]
        batch_results = PipelineInterface(batch)
        results.extend(batch_results)
    return results

def main():
    config = load_config()
    batch_size = 32  # 可以根据GPU内存调整
    
    for model_name, model_path in config['model_path'].items():
        print(f"Processing model: {model_name}")
        model, tokenizer, device = load_model(model_path)
        
        # 设置模型为评估模式
        model.eval()
        
        PipelineInterface = TextClassificationPipeline(
            model=model, 
            tokenizer=tokenizer, 
            top_k=None,
            device=device if device.type == 'cuda' else -1,  # 指定GPU设备
            batch_size=batch_size,
            truncation=True,  # 启用截断
            max_length=512,   # 设置最大长度
            padding=True      # 启用填充
        )
# ----------------------------------改不同的函数和输入数据源------------------------
        df = load_data(config['data_path']['CSI_4000'])
        
        # 使用tqdm显示进度
        texts = df['text'].tolist()
        results = []
        
        # 批量处理文本
        for i in tqdm(range(0, len(texts), batch_size), desc=f"Processing {model_name}"):
            batch_texts = texts[i:i + batch_size]
            batch_results = PipelineInterface(batch_texts)
            results.extend(batch_results)
        
        # 更新DataFrame
        for index, result in enumerate(results):
            score = get_score([result])  # 注意这里需要将单个结果包装成列表
            df.loc[index, 'pos'] = score['Positive']
            df.loc[index, 'neg'] = score['Negative']
            df.loc[index, 'neu'] = score.get('Neutral', np.nan)
        
        # 保存结果
# ----------------------------------改不同的保存数据源------------------------
        df.to_csv(f'{config["save_path"]['CSI_4000']}_{model_name}.csv', index=False)
        
        # 清理GPU内存
        if device.type == 'cuda':
            torch.cuda.empty_cache()
            
if __name__ == '__main__':
    main()
