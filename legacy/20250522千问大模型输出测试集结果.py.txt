import torch
from transformers import AutoModelForCausalLM, AutoTokenizer
import pandas as pd
import numpy as np
from sklearn.metrics import accuracy_score, precision_recall_fscore_support
import logging
from tqdm import tqdm
import os
from transformers import BitsAndBytesConfig

# 设置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('evaluation.log'),
        logging.StreamHandler()
    ]
)

def load_model_and_tokenizer():
    """加载微调后的模型和分词器"""
    logging.info("开始加载模型和分词器...")
    model_path = "./qwen_finetuned"
    
    tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
    tokenizer.padding_side = 'left'
    
    # 设置4bit量化的计算数据类型为float16
    quantization_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True,
        bnb_4bit_quant_type="nf4"
    )
    
    model = AutoModelForCausalLM.from_pretrained(
        model_path,
        quantization_config=quantization_config,
        torch_dtype=torch.float16,
        device_map="auto",
        trust_remote_code=True
    )
    
    return model, tokenizer

def load_test_data(data_path):
    """加载测试数据，使用与微调时相同的划分方式"""
    logging.info("加载测试数据...")
    df = pd.read_excel(data_path)
    
    # 确保数据列存在
    if 'text' not in df.columns or 'label' not in df.columns:
        raise ValueError("数据集中缺少必要的列")
    
    # 处理标签 - 保持原始数字标签
    df['label'] = df['label'].astype(int)
    
    # 使用与微调时相同的随机种子和划分方式
    train_df = df.sample(n=min(500, len(df)), random_state=42)
    test_df = df.drop(train_df.index)
    
    logging.info(f"测试集大小: {len(test_df)}")
    logging.info(f"测试集标签分布:\n{test_df['label'].value_counts()}")
    
    return test_df

def get_sentiment_probabilities_batch(texts, model, tokenizer, batch_size=8):
    """批量获取情感标签的概率分布"""
    all_probs = []
    
    # 将文本分批处理
    for i in range(0, len(texts), batch_size):
        batch_texts = texts[i:i + batch_size]
        
        # 构建批量prompt
        prompts = [
            f"请分析以下文本的情感倾向（积极/中性/消极）：\n{text}\n情感倾向："
            for text in batch_texts
        ]
        
        # 批量编码输入
        inputs = tokenizer(
            prompts,
            padding=True,
            truncation=True,
            return_tensors="pt"
        ).to(model.device)
        
        # 批量生成输出
        with torch.no_grad():
            outputs = model.generate(
                **inputs,
                max_new_tokens=10,
                num_return_sequences=1,
                temperature=0.7,
                do_sample=True
            )
        
        # 处理每个输出
        for output in outputs:
            response = tokenizer.decode(output, skip_special_tokens=True)
            sentiment = response.split("情感倾向：")[-1].strip()
            
            # 将输出转换为概率分布
            probs = {
                1: 0.0,  # 积极
                0: 0.0,  # 中性
                -1: 0.0  # 消极
            }
            
            if "积极" in sentiment:
                probs[1] = 1.0
            elif "消极" in sentiment:
                probs[-1] = 1.0
            else:
                probs[0] = 1.0
            
            all_probs.append(probs)
    
    return all_probs

def evaluate_model(model, tokenizer, test_df, batch_size=8):
    """评估模型性能"""
    logging.info("开始评估模型...")
    
    predictions = []
    true_labels = []
    prediction_probs = []  # 存储预测概率
    
    # 准备批量数据
    texts = test_df['text'].tolist()
    true_labels = test_df['label'].tolist()
    
    # 使用tqdm显示进度
    progress_bar = tqdm(
        range(0, len(texts), batch_size),
        desc="评估进度",
        unit="批次",
        ncols=100,
        bar_format="{l_bar}{bar}| {n_fmt}/{total_fmt} [{elapsed}<{remaining}, {rate_fmt}]"
    )
    
    # 批量处理
    for i in progress_bar:
        batch_texts = texts[i:i + batch_size]
        batch_true_labels = true_labels[i:i + batch_size]
         
        batch_probs = get_sentiment_probabilities_batch(batch_texts, model, tokenizer, batch_size)
        
        # 处理每个预测结果
        for probs, true_label in zip(batch_probs, batch_true_labels):
            pred_label = max(probs.items(), key=lambda x: x[1])[0]
            predictions.append(pred_label)
            prediction_probs.append(probs)  # 保存预测概率
        
        # 更新进度条描述
        progress_bar.set_postfix({
            "当前批次": f"{i//batch_size + 1}/{(len(texts) + batch_size - 1)//batch_size}"
        })
    
    # 计算评估指标
    accuracy = accuracy_score(true_labels, predictions)
    precision, recall, f1, _ = precision_recall_fscore_support(
        true_labels, 
        predictions, 
        average='weighted'
    )
    
    # 打印详细结果
    logging.info(f"准确率: {accuracy:.4f}")
    logging.info(f"精确率: {precision:.4f}")
    logging.info(f"召回率: {recall:.4f}")
    logging.info(f"F1分数: {f1:.4f}")
    
    # 打印混淆矩阵
    from sklearn.metrics import confusion_matrix
    cm = confusion_matrix(true_labels, predictions, labels=[-1, 0, 1])
    logging.info("\n混淆矩阵:")
    logging.info(cm)
    
    # 打印每个类别的详细指标
    class_precision, class_recall, class_f1, _ = precision_recall_fscore_support(
        true_labels, 
        predictions, 
        average=None,
        labels=[-1, 0, 1]
    )
    
    logging.info("\n各类别详细指标:")
    for i, label in enumerate([-1, 0, 1]):
        label_name = "消极" if label == -1 else "中性" if label == 0 else "积极"
        logging.info(f"{label_name}:")
        logging.info(f"  精确率: {class_precision[i]:.4f}")
        logging.info(f"  召回率: {class_recall[i]:.4f}")
        logging.info(f"  F1分数: {class_f1[i]:.4f}")
    
    # 保存预测结果
    results_df = pd.DataFrame({
        'text': texts,
        'score': true_labels,
        'pos': [prob[1] for prob in prediction_probs],  # 积极概率
        'neg': [prob[-1] for prob in prediction_probs],  # 消极概率
        'neu': [prob[0] for prob in prediction_probs]   # 中性概率
    })
    
    # 确保保存目录存在
    save_dir = r"F:\论文\投资者论坛\result\20250522千问大模型测试结果对比"
    os.makedirs(save_dir, exist_ok=True)
    
    # 保存结果
    results_df.to_csv(os.path.join(save_dir, 'prediction_results.csv'), index=False, encoding='utf-8')
    logging.info(f"预测结果已保存至: {os.path.join(save_dir, 'prediction_results.csv')}")
    
    return {
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall,
        'f1': f1,
        'class_metrics': {
            label: {
                'precision': p,
                'recall': r,
                'f1': f
            }
            for label, p, r, f in zip(
                [-1, 0, 1],
                class_precision,
                class_recall,
                class_f1
            )
        }
    }

def main():
    try:
        # 加载模型和分词器
        model, tokenizer = load_model_and_tokenizer()
        
        # 加载测试数据
        test_df = load_test_data("F:/论文/投资者论坛/dataset/text_data/20250521台式预训练600条.xlsx")
        
        # 评估模型，设置批处理大小为8
        results = evaluate_model(model, tokenizer, test_df, batch_size=8)
        
        # 保存评估指标
        save_dir = r"F:\论文\投资者论坛\result\20250522千问大模型测试结果对比"
        os.makedirs(save_dir, exist_ok=True)
        
        # 保存总体评估结果
        pd.DataFrame([results]).to_csv(os.path.join(save_dir, 'evaluation_results.csv'), index=False)
        
        # 保存详细评估结果
        detailed_results = {
            'overall': {
                'accuracy': results['accuracy'],
                'precision': results['precision'],
                'recall': results['recall'],
                'f1': results['f1']
            },
            'class_metrics': results['class_metrics']
        }
        pd.DataFrame(detailed_results).to_csv(os.path.join(save_dir, 'detailed_evaluation_results.csv'), index=False)
        
    except Exception as e:
        logging.error(f"评估过程中出现错误: {str(e)}")
        import traceback
        logging.error(traceback.format_exc())
        raise e

if __name__ == "__main__":
    main()
