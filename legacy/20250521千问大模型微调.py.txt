# 设置设备
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
logging.info(f"使用设备: {device}")

def load_data(data_path):
    """加载并预处理数据"""
    logging.info("开始加载数据...")
    try:
        df = pd.read_excel(data_path)
        logging.info(f"成功读取Excel文件，列名: {df.columns.tolist()}")
    except Exception as e:
        logging.error(f"读取Excel文件失败: {str(e)}")
        raise
    
    # 确保数据列存在
    if 'text' not in df.columns:
        logging.error(f"数据集中没有'text'列。可用列: {df.columns.tolist()}")
        raise ValueError(f"数据集中没有'text'列。可用列: {df.columns.tolist()}")
        
    if 'label' not in df.columns:
        logging.error(f"数据集中没有'label'列。可用列: {df.columns.tolist()}")
        raise ValueError(f"数据集中没有'label'列。可用列: {df.columns.tolist()}")
    
    # 处理空值
    logging.info(f"原始数据大小: {len(df)}")
    df = df.dropna(subset=['text'])  # 删除text列的空值
    logging.info(f"删除空值后数据大小: {len(df)}")
    
    # 从label为1、0、-1的标签中各随机抽取n=500条样本混在一起，打乱顺序，按照8：2的比例分割训练集和测试集
    # 抽取n=500的原因是在现有数据集中正例样本至多有500条，保证每个标签的样本是均衡的
    label_pos_df = df[df['label'] == 1].sample(n=500, random_state=42)
    label_neutral_df = df[df['label'] == 0].sample(n=500, random_state=42)
    label_neg_df = df[df['label'] == -1].sample(n=500, random_state=42)
    
    # 合并样本
    df = pd.concat([label_pos_df, label_neutral_df, label_neg_df])

    # 方法2：使用sklearn的train_test_split（推荐）
    train_df, test_df = train_test_split(df, test_size=0.2, random_state=42)
  
    # 转换为模型输入格式
    train_texts = train_df['text'].tolist()
    test_texts = test_df['text'].tolist()
    
    # 简化的数据集结构
    train_dataset = Dataset.from_dict({"text": train_texts})
    test_dataset = Dataset.from_dict({"text": test_texts})
    
    logging.info(f"数据加载完成。训练集大小: {len(train_dataset)}, 测试集大小: {len(test_dataset)}")
    
    return train_dataset, test_dataset

def load_model_and_tokenizer():
    """加载模型和分词器"""
    logging.info("开始加载模型和分词器...")
    model_name = "Qwen/Qwen2.5-7B-Instruct"
    
    # 加载分词器
    tokenizer = AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)
    
    # 使用4位量化加载模型，大幅减少显存使用
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        load_in_4bit=True,    # 使用4位量化
        torch_dtype=torch.float16,
        device_map="auto",    # 重新启用device_map
        trust_remote_code=True,
        low_cpu_mem_usage=True
    )
    
    # 配置LoRA，使用更小的r值以减少内存使用
    lora_config = LoraConfig(
        r=8,              # 减小LoRA秩以节省显存
        lora_alpha=16,
        target_modules=["q_proj", "k_proj", "v_proj", "o_proj"],
        lora_dropout=0.05,
        bias="none",
        task_type="CAUSAL_LM"
    )
    
    # 准备模型进行LoRA训练
    model = prepare_model_for_kbit_training(model)
    model = get_peft_model(model, lora_config)
    
    logging.info("模型和分词器加载完成")
    return model, tokenizer

def collate_fn(batch, tokenizer):
    """数据整理函数"""
    # 根据批次结构获取文本
    if len(batch) > 0 and isinstance(batch[0], dict) and 'text' in batch[0]:
        texts = [item['text'] for item in batch]
    else:
        texts = batch
    
    # 对文本进行编码
    encodings = tokenizer(
        texts,
        padding="max_length",
        truncation=True,
        max_length=256,
        return_tensors="pt"
    )
    
    # 将数据移动到GPU
    input_ids = encodings['input_ids'].to(device)
    attention_mask = encodings['attention_mask'].to(device)
    
    # 为微调设置标签，使用输入ID作为标签（自回归方式）
    labels = input_ids.clone()
    
    return {
        'input_ids': input_ids,
        'attention_mask': attention_mask,
        'labels': labels
    }

# 自定义Trainer子类，覆盖_move_model_to_device方法
class CustomTrainer(Trainer):
    def _move_model_to_device(self, model, device):
        # 不执行任何操作，因为模型已经在正确的设备上
        self.model = model
        return model

def compute_metrics(eval_preds):
    """简化的评估指标计算 - 仅返回损失"""
    try:
        logits, labels = eval_preds
        
        # 计算损失
        loss = torch.nn.functional.cross_entropy(
            torch.tensor(logits.reshape(-1, logits.shape[-1])), 
            torch.tensor(labels.reshape(-1)),
            ignore_index=-100
        ).item()
        
        # 计算困惑度
        perplexity = np.exp(loss)
        
        return {
            "loss": float(loss),
            "perplexity": float(perplexity)
        }
    except Exception as e:
        logging.error(f"计算指标时出错: {str(e)}")
        return {"loss": 0.0, "perplexity": 0.0}

def train_model(model, tokenizer, train_dataset, test_dataset):
    """训练模型"""
    logging.info("开始训练模型...")
    start_time = time.time()
    
    # 训练参数 - 优化显存使用
    training_args = TrainingArguments(
        output_dir="./qwen_finetuned",
        num_train_epochs=3,
        per_device_train_batch_size=1,     # 减小到1以节省显存
        per_device_eval_batch_size=1,      # 减小到1以节省显存
        gradient_accumulation_steps=16,    # 增加梯度累积步数以保持有效批次大小
        learning_rate=2e-5,
        weight_decay=0.01,
        logging_dir='./logs',
        logging_steps=10,
        save_steps=50,
        eval_steps=50,
        fp16=True,
        # 显存优化参数
        gradient_checkpointing=True,       # 启用梯度检查点以节省显存
        optim="adamw_torch",              # 使用PyTorch的AdamW优化器
        max_grad_norm=1.0,                # 使用梯度裁剪
        # 确保使用GPU
        no_cuda=False,                    # 允许使用CUDA
        dataloader_drop_last=True,
        # 关键：不要移除未使用的列
        remove_unused_columns=False,
    )
    
    # 定义一个预处理函数，直接在训练前处理数据
    def preprocess_function(examples):
        # 对文本进行编码
        encodings = tokenizer(
            examples["text"],
            padding="max_length",
            truncation=True,
            max_length=128,               # 进一步减小序列长度以节省显存
        )
        
        result = {}
        # 不使用张量格式返回，让Dataset.map处理为普通列表
        result["input_ids"] = encodings["input_ids"]
        result["attention_mask"] = encodings["attention_mask"]
        result["labels"] = encodings["input_ids"].copy()  # 使用复制而不是clone
        
        return result
    
    # 预处理数据集
    logging.info("预处理训练数据集...")
    train_dataset = train_dataset.map(preprocess_function, batched=True)
    logging.info("预处理测试数据集...")
    test_dataset = test_dataset.map(preprocess_function, batched=True)
    
    # 使用自定义Trainer
    trainer = CustomTrainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
        eval_dataset=test_dataset,
        compute_metrics=compute_metrics,
    )
    
    # 开始训练
    trainer.train()
    
    # 保存模型
    trainer.save_model()
    tokenizer.save_pretrained("./qwen_finetuned_4000")
    
    end_time = time.time()
    training_time = end_time - start_time
    logging.info(f"训练完成！总耗时: {training_time/60:.2f}分钟")
    
    return trainer
