import torch
from transformers import BertForSequenceClassification, BertTokenizer, TrainingArguments, Trainer, TrainerCallback
from datasets import Dataset
import pandas as pd
import numpy as np
from sklearn.metrics import accuracy_score, precision_recall_fscore_support
import os
import time
from datetime import datetime
import logging
from tqdm import tqdm
from sklearn.model_selection import train_test_split

# 设置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('macbert_training.log'),
        logging.StreamHandler()
    ]
)

# 设置设备
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
logging.info(f"使用设备: {device}")

def load_data(data_path):
    """加载并预处理数据"""
    logging.info("开始加载数据...")
    try:
        df = pd.read_excel(data_path)
        logging.info(f"成功读取Excel文件，列名: {df.columns.tolist()}")
    except Exception as e:
        logging.error(f"读取Excel文件失败: {str(e)}")
        raise
    
    # 确保数据列存在
    required_columns = ['text', 'label']
    missing_columns = [col for col in required_columns if col not in df.columns]
    if missing_columns:
        error_msg = f"数据集中缺少必要的列: {missing_columns}。可用列: {df.columns.tolist()}"
        logging.error(error_msg)
        raise ValueError(error_msg)
    
    # 处理空值
    logging.info(f"原始数据大小: {len(df)}")
    df = df.dropna(subset=['text'])  # 删除text列的空值
    logging.info(f"删除空值后数据大小: {len(df)}")
    
    # 检查标签分布
    label_counts = df['label'].value_counts()
    logging.info(f"标签分布:\n{label_counts}")
    
    # 从每个标签类别中采样
    sample_size = 500
    sampled_dfs = []
    for label in [-1, 0, 1]:
        label_df = df[df['label'] == label]
        if len(label_df) < sample_size:
            logging.warning(f"标签 {label} 的样本数量不足 {sample_size}，将使用所有可用样本")
            sampled_dfs.append(label_df)
        else:
            sampled_dfs.append(label_df.sample(n=sample_size, random_state=42))
    
    # 合并样本
    df = pd.concat(sampled_dfs)
    
    # 将标签转换为0,1,2格式（MacBERT需要从0开始的连续标签）
    label_map = {-1: 0, 0: 1, 1: 2}
    df['label'] = df['label'].map(label_map)
    
    # 分割训练集和测试集
    train_df, test_df = train_test_split(df, test_size=0.2, random_state=42, stratify=df['label'])
    
    # 转换为模型输入格式
    train_dataset = Dataset.from_dict({
        "text": train_df['text'].tolist(),
        "label": train_df['label'].tolist()
    })
    test_dataset = Dataset.from_dict({
        "text": test_df['text'].tolist(),
        "label": test_df['label'].tolist()
    })
    
    test_df.to_csv("F:/论文/投资者论坛/dataset/text_data/verify_data/沪深300股票问答数据-4000条-test.csv", index=False)

    logging.info(f"数据加载完成。训练集大小: {len(train_dataset)}, 测试集大小: {len(test_dataset)}")
    logging.info(f"训练集标签分布: {pd.Series(train_df['label']).value_counts().to_dict()}")
    logging.info(f"测试集标签分布: {pd.Series(test_df['label']).value_counts().to_dict()}")
    
    return train_dataset, test_dataset

def load_model_and_tokenizer():
    """加载模型和分词器"""
    logging.info("开始加载模型和分词器...")
    model_name = "hfl/chinese-macbert-base"
    
    try:
        # 加载分词器
        tokenizer = BertTokenizer.from_pretrained(model_name)
        
        # 加载模型，设置为3分类任务
        model = BertForSequenceClassification.from_pretrained(
            model_name,
            num_labels=3,
            problem_type="single_label_classification"
        )
        
        model = model.to(device)
        logging.info("模型和分词器加载完成")
        return model, tokenizer
    except Exception as e:
        logging.error(f"加载模型或分词器时出错: {str(e)}")
        raise

def compute_metrics(eval_preds):
    """计算评估指标"""
    try:
        logits, labels = eval_preds
        predictions = np.argmax(logits, axis=-1)
        
        # 计算各项指标
        accuracy = accuracy_score(labels, predictions)
        precision, recall, f1, _ = precision_recall_fscore_support(
            labels, predictions, average='weighted'
        )
        
        return {
            "accuracy": accuracy,
            "precision": precision,
            "recall": recall,
            "f1": f1
        }
    except Exception as e:
        logging.error(f"计算评估指标时出错: {str(e)}")
        return {
            "accuracy": 0.0,
            "precision": 0.0,
            "recall": 0.0,
            "f1": 0.0
        }

class MetricsCallback(TrainerCallback):
    """自定义回调函数，用于记录训练指标"""
    def __init__(self):
        self.best_metrics = {
            "accuracy": 0.0,
            "precision": 0.0,
            "recall": 0.0,
            "f1": 0.0
        }
        
    def on_log(self, args, state, control, logs=None, **kwargs):
        if logs is not None:
            # 更新最佳指标
            for metric in self.best_metrics.keys():
                eval_metric = f"eval_{metric}"
                if eval_metric in logs and logs[eval_metric] > self.best_metrics[metric]:
                    self.best_metrics[metric] = logs[eval_metric]
                    logging.info(f"新的最佳{metric}: {logs[eval_metric]:.4f}")

def train_model(model, tokenizer, train_dataset, test_dataset):
    """训练模型"""
    logging.info("开始训练模型...")
    start_time = time.time()
    
    # 训练参数
    training_args = TrainingArguments(
        output_dir="./macbert_finetuned",
        num_train_epochs=3,
        per_device_train_batch_size=16,
        per_device_eval_batch_size=16,
        learning_rate=2e-5,
        weight_decay=0.01,
        logging_steps=10,
        save_steps=50,
        eval_steps=50,
        fp16=True,
        gradient_accumulation_steps=4
    )
    
    # 定义预处理函数
    def preprocess_function(examples):
        return tokenizer(
            examples["text"],
            padding="max_length",
            truncation=True,
            max_length=512,
            return_tensors="pt"
        )
    
    try:
        # 预处理数据集
        logging.info("预处理训练数据集...")
        train_dataset = train_dataset.map(preprocess_function, batched=True)
        logging.info("预处理测试数据集...")
        test_dataset = test_dataset.map(preprocess_function, batched=True)
        
        # 创建回调函数实例
        metrics_callback = MetricsCallback()
        
        # 创建训练器
        trainer = Trainer(
            model=model,
            args=training_args,
            train_dataset=train_dataset,
            eval_dataset=test_dataset,
            compute_metrics=compute_metrics,
            callbacks=[metrics_callback]
        )
        
        # 开始训练
        trainer.train()
        
        # 保存模型
        trainer.save_model()
        tokenizer.save_pretrained("./macbert_finetuned")
        
        end_time = time.time()
        training_time = end_time - start_time
        logging.info(f"训练完成！总耗时: {training_time/60:.2f}分钟")
        logging.info(f"最佳评估指标: {metrics_callback.best_metrics}")
        
        return trainer
    except Exception as e:
        logging.error(f"训练过程中出错: {str(e)}")
        raise

def main():
    # 记录开始时间
    total_start_time = time.time()
    
    try:
        # 设置种子，确保可重现性
        torch.manual_seed(42)
        np.random.seed(42)
        
        # 显示环境信息
        logging.info(f"CUDA是否可用: {torch.cuda.is_available()}")
        if torch.cuda.is_available():
            logging.info(f"CUDA设备数量: {torch.cuda.device_count()}")
            logging.info(f"当前CUDA设备: {torch.cuda.current_device()}")
            logging.info(f"CUDA设备名称: {torch.cuda.get_device_name(0)}")
            torch.cuda.empty_cache()
            logging.info(f"GPU内存情况: 已分配 {torch.cuda.memory_allocated(0)/1024**3:.2f} GB")
                
        # 加载数据
        logging.info("开始加载数据集...")
        train_dataset, test_dataset = load_data("F:/论文/投资者论坛/dataset/text_data/沪深300股票问答数据-4000条.xlsx")
        
        # 加载模型和分词器
        logging.info("开始加载模型和分词器...")
        model, tokenizer = load_model_and_tokenizer()
        
        # 训练模型
        logging.info("开始训练模型...")
        trainer = train_model(model, tokenizer, train_dataset, test_dataset)
        
        # 评估模型
        logging.info("开始评估模型...")
        eval_results = trainer.evaluate()
        logging.info(f"最终评估结果: {eval_results}")
        
        # 记录总耗时

        total_time = total_end_time - total_start_time
        logging.info(f"整个过程完成！总耗时: {total_time/60:.2f}分钟")
        
    except Exception as e:

if __name__ == "__main__":
    main()
